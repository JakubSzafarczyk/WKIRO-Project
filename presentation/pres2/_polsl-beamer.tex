\begin{frame}[fragile]{Cel projektu i dane}
	\begin{itemize}
		\item Cel: klasyfikacja sekwencji ruchu 3D z danych motion capture.
		\item Dane: Riglet et al. 2024, 30 uczestników, 63 markery, Vicon 100 Hz.
		\item Wejście: współrzędne XYZ markerów w kolejnych klatkach.
		\item Wykorzystano pliki CSV z katalogów \texttt{Post\_Process}.
	\end{itemize}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}[fragile]{Pipeline i model}
	\begin{columns}[T,onlytextwidth]
		\column{0.48\textwidth}
		\textbf{Przetwarzanie danych}
		\begin{itemize}
			\item wczytanie plików CSV,
			\item centrowanie markerów względem miednicy,
			\item standaryzacja cech,
			\item podział na okna czasowe,
			\item podział \texttt{train/val/test}.
		\end{itemize}
		
		\column{0.48\textwidth}
		\textbf{Model LSTM}
		\begin{itemize}
			\item analizuje zależności między klatkami,
			\item wejście: \texttt{(window, n\_features)},
			\item wyjście: klasyfikacja softmax,
			\item agregacja okien do wyniku nagrania przez modę.
		\end{itemize}
	\end{columns}
\end{frame}


%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}[fragile]{Zadania klasyfikacji}
	\begin{itemize}
		\item \textbf{\texttt{gait\_type}} -- rozpoznanie typu ruchu, np. chód wolny, szybki lub bieg.
		\item \textbf{\texttt{sex}} -- rozpoznanie płci uczestnika.
		\item \textbf{\texttt{participant\_id}} -- identyfikacja konkretnego uczestnika.
	\end{itemize}
	
	\vspace{0.8em}
	
	\begin{block}{Najważniejsze założenie}
		Model nie dostaje gotowych cech biomechanicznych -- uczy się wzorców bezpośrednio z sekwencji położeń markerów.
	\end{block}
\end{frame}

	\begin{block}{Skala}
		7 konfiguracji $\times$ 3 zadania = \textbf{21 wytrenowanych modeli}, optymalizator Adam, lr = 0{,}001.
	\end{block}
\end{frame}


%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}[fragile]{Ewaluacja}
	\begin{itemize}
		\item Wyniki liczono na dwóch poziomach:
		\begin{itemize}
			\item \textbf{okno} -- klasyfikacja pojedynczego fragmentu sekwencji,
			\item \textbf{nagranie} -- jedna decyzja po agregacji predykcji okien.
		\end{itemize}
		\item Główna metryka rankingowa: \textbf{F1-macro}.
		\item Dodatkowo analizowano accuracy, precision, recall i macierze pomyłek.
	\end{itemize}
	
	\vspace{0.5em}
	
	\begin{block}{Dlaczego F1-macro?}
		Metryka traktuje klasy równoważnie, więc lepiej pokazuje jakość klasyfikacji przy nierównych licznościach klas.
	\end{block}
\end{frame}

\begin{frame}[fragile]{Interpretacja: możliwy overfitting}
	\begin{itemize}
		\item Wyniki 100\% dla płci i ID uczestnika są bardzo wysokie i wymagają ostrożnej interpretacji.
		\item Podział \texttt{within\_participant} powoduje, że dane tego samego uczestnika występują w treningu i teście.
		\item Model może uczyć się sygnatury uczestnika lub nagrania, np. proporcji ciała, ustawienia markerów albo charakterystycznego wzorca ruchu.
		\item Do mocniejszej weryfikacji potrzebny byłby podział \texttt{participant\_holdout}.
	\end{itemize}
\end{frame}

\begin{frame}[fragile]{Wnioski}
	\begin{itemize}
		\item LSTM skutecznie klasyfikuje sekwencje motion capture na podstawie okien czasowych.
		\item Najtrudniejszym zadaniem było rozpoznawanie typu chodu.
		\item Agregacja okien do poziomu nagrania zwykle poprawiała wynik, ale nie zawsze.
		\item \texttt{w50-L64} był najbardziej uniwersalną konfiguracją.
		\item Kolejny krok: walidacja na nowych uczestnikach i sprawdzenie odporności modeli na zmianę sesji.
	\end{itemize}
\end{frame}
